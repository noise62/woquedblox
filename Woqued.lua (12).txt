local function newI(cls, props, par)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do o[k] = v end
    if par then o.Parent = par end
    return o
end
local function corner(p, r)
    return newI("UICorner", {CornerRadius = UDim.new(0, r or 6)}, p)
end
local function stroke(p, col, th)
    return newI("UIStroke", {
        Color = col or C.border, Thickness = th or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, p)
end
local function tw(o, t, props, es, ed)
    TweenService:Create(o, TweenInfo.new(
        t or 0.15,
        es or Enum.EasingStyle.Quad,
        ed or Enum.EasingDirection.Out
    ), props):Play()
end

local function isMurd(pl)

    if c:FindFirstChild("Gun") then return true end
    local b = pl:FindFirstChild("Backpack")
    return b and b:FindFirstChild("Gun") ~= nil
end

-- ═══════════════════════════════════════════
--     DRAG HELPER (mouse + touch)
-- ═══════════════════════════════════════════
local function makeDraggable(frame, handleFrame)
    local handle = handleFrame or frame
    local d, s, p = false, nil, nil
    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            d = true; s = inp.Position; p = frame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1
        or inp.UserInputType == Enum.UserInputType.Touch then
            d = false
        end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if d and (inp.UserInputType == Enum.UserInputType.MouseMovement
        or inp.UserInputType == Enum.UserInputType.Touch) then
            local delta = inp.Position - s
            frame.Position = UDim2.new(
                p.X.Scale, p.X.Offset + delta.X,
                p.Y.Scale, p.Y.Offset + delta.Y
            )
        end
    end)
end

        n.lbl.Font = Enum.Font.GothamBold

    hdrCatTitle.Text = newCat
    hdrCatDesc.Text  = CAT_DESC[newCat] or ""
    buildModules()
end

local function addTrail(from2D, to2D)
    if not MOD.shoottrails.enabled or not HAS_DRAW then return end
    local t = { startTime = tick(), duration = 0.65, lines = {} }
    local cfg = {
        {col=Color3.fromRGB(255,255,255), thick=1.2, alpha=1.00},
        {col=Color3.fromRGB(255,120,120), thick=3.0, alpha=0.45},
        {col=Color3.fromRGB(255, 60, 60), thick=6.0, alpha=0.18},
        {col=Color3.fromRGB(255, 20, 20), thick=11,  alpha=0.07},
    }
    for _, c in ipairs(cfg) do
        local l = Drawing.new("Line")
        l.From = from2D; l.To = to2D
        l.Color = c.col; l.Thickness = c.thick
        l.Transparency = c.alpha; l.Visible = true
        table.insert(t.lines, l)
    end
    table.insert(S.trails, t)
end
local function updateTrails()
    if not HAS_DRAW then return end
    local now, i = tick(), 1
    local alphas = {1.00, 0.45, 0.18, 0.07}
    while i <= #S.trails do
        local t = S.trails[i]
        local age = now - t.startTime
        if age >= t.duration then
            for _, l in ipairs(t.lines) do l:Remove() end
            table.remove(S.trails, i)
        else
            local fade = 1 - (age / t.duration)
            for j, l in ipairs(t.lines) do
                l.Transparency = alphas[j] * fade
            end
            i = i + 1
        end
    end
end

-- Knife pose: parametric root CFrame + swing angle of each handle.
-- Idle pose is like screenshot 2: blade leans up-left, handles down-right.
local BL1, BL2, TIP_L, HD_L = 0.75, 0.60, 0.42, 0.95
-- roll 55° = blade lies almost horizontally, tip up-left (like CS2)
local rootIdle   = CFrame.new(0.50, -0.12, 0.66) * CFrame.Angles(math.rad(-4), 0, math.rad(55))
local a1Idle     = math.rad(14)
local a2Idle     = math.rad(-8)
local rootVal    = Instance.new("CFrameValue")
local swingVal   = Instance.new("NumberValue")
local swing2Val  = Instance.new("NumberValue")
rootVal.Value = rootIdle; swingVal.Value = a1Idle; swing2Val.Value = a2Idle

local function applyKnifePose()
    local root = rootVal.Value * CFrame.new(S.vmBobX or 0, S.vmBobY or 0, 0)
    vmPivot.CFrame = root
    -- curved blade: three chained segments
    local j0 = root * CFrame.new(0, 0.10, 0)
    vmBlade.CFrame = j0 * CFrame.new(0, BL1/2, 0)
    local j1 = j0 * CFrame.new(0, BL1, 0) * CFrame.Angles(0, 0, math.rad(14))
    vmBlade2.CFrame = j1 * CFrame.new(0, BL2/2, 0)
    local j2 = j1 * CFrame.new(0, BL2, 0) * CFrame.Angles(0, 0, math.rad(16))
    vmTip.CFrame = j2 * CFrame.new(0, TIP_L/2 - 0.04, 0) * CFrame.Angles(0, 0, math.rad(12))
    -- handles swing around the pivot axis
    vmH1.CFrame = root * CFrame.new(-0.115, 0, 0)
                 * CFrame.Angles(swingVal.Value, 0, 0)
                 * CFrame.new(0, -(HD_L/2 + 0.06), 0)
    vmH2.CFrame = root * CFrame.new(0.115, 0, 0)
                 * CFrame.Angles(swing2Val.Value, 0, 0)
                 * CFrame.new(0, -(HD_L/2 + 0.06), 0)
    vmLatch.CFrame = root * CFrame.new(0.115, 0, 0)
                   * CFrame.Angles(swing2Val.Value, 0, 0)
                   * CFrame.new(0, -(HD_L + 0.22), 0)
    -- модель из тулбокса живёт в той же точке (тот же боб/свинг/осмотр)
    if S.vmKnifeModel then
        S.vmKnifeModel:PivotTo(root * KNIFE_ROT)
    end
end
rootVal.Changed:Connect(applyKnifePose)
swingVal.Changed:Connect(applyKnifePose)
swing2Val.Changed:Connect(applyKnifePose)

-- Загрузка бабочки из тулбокса (public domain): первая загрузившаяся
-- заменяет процедурный нож. Чистка от скриптов/звуков, масштаб ~1.9.
task.spawn(function()
    local function tryLoad(id)
        local ok, objs = pcall(game.GetObjects, game, "rbxassetid://" .. tostring(id))
        if not ok or type(objs) ~= "table" or #objs == 0 then return nil end
        for _, o in ipairs(objs) do
            if o:IsA("Model") then return o end
            if o:IsA("BasePart") then
                local m = Instance.new("Model"); o.Parent = m; return m
            end
        end
        return nil
    end
    for _, id in ipairs(KNIFE_IDS) do
        local mdl = tryLoad(id)
        if mdl then
            for _, d in ipairs(mdl:GetDescendants()) do
                if not d:IsA("BasePart") and not d:IsA("SpecialMesh")
                   and not d:IsA("BlockMesh") and not d:IsA("CylinderMesh")
                   and not d:IsA("Attachment") then
                    d:Destroy()
                end
            end
            for _, d in ipairs(mdl:GetDescendants()) do
                if d:IsA("BasePart") then
                    d.Anchored = true
                    d.CanCollide = false; d.CanTouch = false; d.CanQuery = false
                    d.CastShadow = false
                end
            end
            mdl.Parent = vmModel
            local ext = mdl:GetExtentsSize()
            local maxDim = math.max(ext.X, ext.Y, ext.Z)
            if maxDim > 0 then mdl:ScaleTo(1.9 / maxDim) end
            S.vmKnifeModel = mdl
            S.knifeExternal = true
            for _, p in ipairs(knifeParts) do p.Transparency = 1 end
            break
        end
    end
end)

local function updateVMBob(dt)
    local hrp = getHRP(); if not hrp then return end
    local spd = hrp.Velocity.Magnitude

    -- weapon sway: lags behind camera turns (like CS viewmodels)
    local look = Cam.CFrame.LookVector
    local yaw = math.atan2(look.X, look.Z)
    if S.lastYaw then
        local d = yaw - S.lastYaw
        if d >  math.pi then d = d - math.pi * 2 end
        if d < -math.pi then d = d + math.pi * 2 end
        if not S.inspecting then
            local target = math.clamp(d * 1.6, -0.055, 0.055)
            S.swayX = S.swayX + (target - S.swayX) * math.min(dt * 10, 1)
        else
            S.swayX = S.swayX * math.max(1 - dt * 10, 0)
        end
    end
    S.lastYaw = yaw

    if spd > 2 and not S.inspecting then
        S.vmBobT = S.vmBobT + dt * 8
        S.vmBobX = math.sin(S.vmBobT) * 0.018 + S.swayX
        S.vmBobY = -math.abs(math.sin(S.vmBobT * 0.5)) * 0.016
    else
        S.vmBobT = 0
        S.vmBobX = S.swayX
        S.vmBobY = 0
    end
    if not S.inspecting then setArmsIdle() end
    applyKnifePose()
    applyAwpPose()
    local pitch = math.asin(math.clamp(Cam.CFrame.UpVector:Dot(Vector3.new(0,0,-1)),-1,1))
    vmCam.CFrame = CFrame.new(0,0,3) * CFrame.Angles(pitch * 0.12, 0, 0)
end

-- Inspect animation (F / knife button):
-- knife = raise with overshoot, full handle flip, side wrist roll,
--         second handle flip while rolled, then back to idle
-- AWP   = lift, 180° roll around the barrel, back
doInspect = function()
    if S.inspecting or not MOD.cspov.enabled then return end
    S.inspecting = true
    local TI = TweenInfo.new

    if S.activeVM == "awp" then
        local raise = awpIdle * CFrame.new(0, 0.18, -0.15)
                    * CFrame.Angles(math.rad(8), math.rad(-6), math.rad(14))

            {Value = raise}):Play()
        TweenService:Create(vmRA, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {CFrame = CFrame.lookAt(Vector3.new(0.75, -1.05, 1.85), Vector3.new(0.40, -0.35, 1.35))}):Play()
        task.delay(0.35, function()
            TweenService:Create(awpVal, TI(0.45, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
                {Value = raise * CFrame.Angles(math.rad(180), 0, 0)}):Play()
        end)
        task.delay(0.85, function()
            TweenService:Create(awpVal, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                {Value = awpIdle}):Play()
            TweenService:Create(vmRA, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                {CFrame = CFrame.lookAt(armPoses.awp.RA.center, armPoses.awp.RA.hand)}):Play()
        end)
        task.delay(1.20, function() S.inspecting = false end)
        return
    end

    local raise = CFrame.new(0.08, -0.05, 0.60) * CFrame.Angles(math.rad(-4), 0, math.rad(24))
    -- 1) raise the knife to the middle of the screen (slight overshoot)
    TweenService:Create(rootVal, TI(0.30, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
        {Value = raise}):Play()
    TweenService:Create(vmRA, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {CFrame = CFrame.lookAt(Vector3.new(0.45, -1.05, 1.35), Vector3.new(0.10, -0.55, 0.62))}):Play()
    TweenService:Create(vmLA, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {CFrame = CFrame.lookAt(Vector3.new(-0.65, -1.30, 1.45), Vector3.new(-0.30, -0.75, 0.70))}):Play()
    -- 2) full flip of the first handle around the pivot
    task.delay(0.34, function()
        TweenService:Create(swingVal, TI(0.50, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
            {Value = a1Idle + math.pi * 2}):Play()
    end)
    -- 3) wrist roll: knife goes sideways
    task.delay(0.90, function()
        TweenService:Create(rootVal, TI(0.24, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
            {Value = raise * CFrame.Angles(0, 0, math.rad(-95))}):Play()
    end)
    -- 4) full flip of the second handle while rolled
    task.delay(1.18, function()
        TweenService:Create(swing2Val, TI(0.46, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
            {Value = a2Idle + math.pi * 2}):Play()
    end)
    -- 5) back to idle
    task.delay(1.70, function()
        swingVal.Value  = a1Idle
        swing2Val.Value = a2Idle
        TweenService:Create(rootVal, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Value = rootIdle}):Play()
        TweenService:Create(vmRA, TI(0.30, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {CFrame = CFrame.lookAt(armPoses.knife.RA.center, armPoses.knife.RA.hand)}):Play()

            {CFrame = CFrame.lookAt(armPoses.knife.LA.center, armPoses.knife.LA.hand)}):Play()
    end)
    task.delay(2.05, function() S.inspecting = false end)
end

--           SILENT AIM  HOOK
-- ═══════════════════════════════════════════
local function installSilentAim()
    local ok = pcall(function()
        local mt = getrawmetatable(game)
        setreadonly(mt, false)
        local oldNC = mt.__namecall
        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if (MOD.silentaim.enabled or MOD.autoshoot.enabled) and S.saTarget then
                if MOD.antiaim.enabled then
                    S.aaPaused = true
                    task.delay(0.22, function() S.aaPaused = false end)
                end
                if method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" then
                    local args = {...}
                    if typeof(args[1]) == "Ray" then
                        local newDir = (S.saTarget.Position - args[1].Origin).Unit * 1000
                        args[1] = Ray.new(args[1].Origin, newDir)
                        return oldNC(self, unpack(args))
                    end
                elseif method == "Raycast" then
                    local args = {...}
                    if typeof(args[1]) == "Vector3" then
                        args[2] = S.saTarget.Position - args[1]
                        return oldNC(self, unpack(args))
                    end
                end
            end
            return oldNC(self, ...)
        end)
    end)
    if not ok then warn("[Woqued] SilentAim hook unavailable") end
end
installSilentAim()

                        addTrail(Vector2.new(s2d.X, s2d.Y), Vector2.new(e2d.X, e2d.Y))

-- ═══════════════════════════════════════════
--     EMOTE MODULE  (Rampage Dance 🔥)
--   Локальный танец поверх ходьбы (Action-приоритет).
-- ═══════════════════════════════════════════
local function stopEmote()
    if S.emoteTrack then
        pcall(function() S.emoteTrack:Stop(0.15) end)
        S.emoteTrack = nil
    end
end

local function playEmoteTrack(animator, animId)
    local track
    local ok = pcall(function()
        local a = Instance.new("Animation")
        a.AnimationId = "rbxassetid://" .. tostring(animId)
        track = animator:LoadAnimation(a)
    end)
    if not ok or not track then return false end
    track.Priority = Enum.AnimationPriority.Action  -- играет даже в движении
    track.Looped = true
    S.emoteTrack = track
    track:Play(0.15)
    return true
end

local function startEmote()
    local char = getChar(); if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid"); if not hum then return end
    local animator = hum:FindFirstChildOfClass("Animator")
    if not animator then
        animator = Instance.new("Animator"); animator.Parent = hum
    end
    S.emoteLastTry = tick()
    task.spawn(function()
        -- основная эмоция из share-ссылки
        if not playEmoteTrack(animator, EMOTE_ID) then S.emoteDead = true return end
        local t0 = tick()
        while S.emoteTrack and not S.emoteTrack.Loaded and tick() - t0 < 3 do
            task.wait(0.1)
        end
        if not S.emoteTrack or S.emoteTrack.Loaded then return end
        -- эмоция не загрузилась (не куплена) → бесплатный запасной танец
        pcall(function() S.emoteTrack:Stop(0) end)
        S.emoteTrack = nil
        if not playEmoteTrack(animator, EMOTE_FALLBACK) then S.emoteDead = true return end
        t0 = tick()
        while S.emoteTrack and not S.emoteTrack.Loaded and tick() - t0 < 3 do
            task.wait(0.1)
        end
        if S.emoteTrack and not S.emoteTrack.Loaded then
            S.emoteDead = true
            stopEmote()
        end
    end)
end

-- ═══════════════════════════════════════════
local function getAaJoints(char)
    return char:FindFirstChild("Neck", true), char:FindFirstChild("Waist", true)
end

local function applyAaHeadDown(char)
    local neck, waist = getAaJoints(char)
    if neck and neck:IsA("Motor6D") then
        if not S.aaNeckC0 then S.aaNeckC0 = neck.C0 end
        neck.C0 = S.aaNeckC0 * CFrame.Angles(math.rad(-85), 0, 0)
    end
    if waist and waist:IsA("Motor6D") then
        if not S.aaWaistC0 then S.aaWaistC0 = waist.C0 end
        waist.C0 = S.aaWaistC0 * CFrame.Angles(math.rad(-55), 0, 0)
    end
end

local function restoreAaPose()
    local char = getChar(); if not char then return end
    local neck, waist = getAaJoints(char)
    if neck and S.aaNeckC0 then neck.C0 = S.aaNeckC0 end
    if waist and S.aaWaistC0 then waist.C0 = S.aaWaistC0 end
    S.aaNeckC0, S.aaWaistC0 = nil, nil
end

onToggle = function(m)
    updateBindsHUD()
    if m.id == "cspov" then
        if m.enabled then
            LP.CameraMaxZoomDistance = 0.5
            LP.CameraMinZoomDistance = 0.5
            vmFrame.Visible = true
            inspectBtn.Visible = true
            setActiveWeapon(MOD.antiaim.enabled and not MOD.emote.enabled
                                and "awp" or "knife")
        else
            LP.CameraMaxZoomDistance = 128
            LP.CameraMinZoomDistance = 0.5
            vmFrame.Visible = false; setVMIdle()
            inspectBtn.Visible = false
        end
    elseif m.id == "watermark" then
        wmFrame.Visible = m.enabled
    elseif m.id == "bindshud" then
        bindsFrame.Visible = m.enabled
        if m.enabled then updateBindsHUD() end
    elseif m.id == "speedhud" then
        speedFrame.Visible = m.enabled
    elseif m.id == "antiaim" then
        if m.enabled then
            if not MOD.emote.enabled then setActiveWeapon("awp") end
        else
            S.aaAngle = 0; S.aaPaused = false
            restoreAaPose()
            setActiveWeapon("knife")
        end
        applyKnifePose(); applyAwpPose()
        if not S.inspecting then setArmsIdle() end
    elseif m.id == "bunnyhop" then
        if not m.enabled and S.bhopBV then
            S.bhopBV:Destroy(); S.bhopBV = nil
        end
    elseif m.id == "emote" then
        if m.enabled then
            S.emoteDead = false
            startEmote()
            -- при эмоции AntiAim остаётся только кручение: без AWP и наклона
            if MOD.antiaim.enabled then setActiveWeapon("knife") end
        else
            stopEmote()
            if MOD.antiaim.enabled then setActiveWeapon("awp") end
        end
    elseif m.id == "esp" then
        if not m.enabled then
            for _, d in pairs(espData) do
                if d.box  then d.box.Visible  = false end
                if d.name then d.name.Visible = false end
            end
        end
    elseif m.id == "silentaim" then
        if fovCircle then fovCircle.Visible = m.enabled and m.set.showFov end
    elseif m.id == "shoottrails" then
        if not m.enabled then
            for _, t in ipairs(S.trails) do
                for _, l in ipairs(t.lines) do l:Remove() end
            end
            S.trails = {}
        end
    end
end

toggleMenu = function()
    if S.animating then return end

        local sz  = win.Size

        local cx  = pos.X.Offset + sz.X.Offset * 0.5

        local ns  = UDim2.new(0, sz.X.Offset * 0.88, 0, sz.Y.Offset * 0.88)

            TweenInfo.new(0.17, Enum.EasingStyle.Quad, Enum.EasingDirection.In),

            win.Visible = false
            win.Size = UDim2.new(0, WW, 0, WH)
            win.Position = WIN_POS

        end)

            TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            {Size = UDim2.new(0, WW, 0, WH), Position = WIN_POS}
        )
        t:Play()
        t.Completed:Connect(function()
            S.open = true; S.animating = false
        end)
    end
end

-- ═══════════════════════════════════════════
--            MAIN  RENDER  LOOP
-- ═══════════════════════════════════════════
RunService.RenderStepped:Connect(function(dt)
    -- FPS counter
    S.fpsCount = S.fpsCount + 1
    if tick() - S.fpsTimer >= 1 then
        S.fps = S.fpsCount; S.fpsCount = 0; S.fpsTimer = tick()
    end

    local char = getChar()
    local hrp  = getHRP()
    local hum  = getHum()

    -- ── AntiAim (upright spin + head bent down, local) ─
    if MOD.antiaim.enabled and hrp then
        if not S.aaPaused then
            S.aaAngle = (S.aaAngle + 0.055) % (math.pi * 2)
        end
        -- yaw-only spin: pitching the HRP pushes the body into walls/ground
        -- and physics flings the player across the map, so the downlook is
        -- done purely via the neck/waist joints instead
        hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, S.aaAngle, 0)
        if MOD.emote.enabled then
            -- эмоция вытесняет наклон вниз: разогнуть суставы и крутиться
            if char and S.aaNeckC0 then restoreAaPose() end
        elseif char then
            applyAaHeadDown(char)
        end
        -- safety: kill physics fling velocity (bhop max is 60, falls < 200)
        local v = hrp.AssemblyLinearVelocity
        if v.Magnitude > 500 then
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
    end

    -- ── Emote: автоперезапуск после респавна / обрыва ─
    if MOD.emote.enabled and not S.emoteDead
       and tick() - (S.emoteLastTry or 0) > 3 then
        local t = S.emoteTrack
        if not t or not t.IsPlaying then startEmote() end
    end

    -- ── BunnyHop (no speed reset on landing, capped at 60) ─
    if MOD.bunnyhop.enabled and hum and hrp then
        local onGround = hum.FloorMaterial ~= Enum.Material.Air
        local vel = hrp.AssemblyLinearVelocity
        local hSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
        local moveDir = hum.MoveDirection

        if not onGround then
            -- airborne: remember speed (max 60) and keep accelerating
            S.bhopAir = true
            if hSpeed > (S.bhopSpeed or 0) then
                S.bhopSpeed = math.min(hSpeed, 60)
            end
            if moveDir.Magnitude > 0 then
                if not S.bhopBV then
                    S.bhopBV = Instance.new("BodyVelocity")
                    S.bhopBV.MaxForce = Vector3.new(1e5, 0, 1e5)
                    S.bhopBV.P = 1e5
                    S.bhopBV.Parent = hrp
                end
                local target = math.min(hSpeed + 1.4, 60)
                S.bhopBV.Velocity = moveDir * target
            else
                if S.bhopBV then
                    S.bhopBV:Destroy(); S.bhopBV = nil
                end
            end
        else
            if S.bhopAir then
                S.bhopAir = false
                S.bhopLandT = tick()
            end
            -- right after landing: jump instantly and keep pushing at the
            -- stored speed, so ground friction cannot reset it
            local justLanded = (tick() - (S.bhopLandT or 0)) < 0.20
            if moveDir.Magnitude > 0 and (S.bhopSpeed or 0) > 17 and justLanded then
                if not S.bhopBV then
                    S.bhopBV = Instance.new("BodyVelocity")
                    S.bhopBV.MaxForce = Vector3.new(1e5, 0, 1e5)
                    S.bhopBV.P = 1e5
                    S.bhopBV.Parent = hrp
                end
                S.bhopBV.Velocity = moveDir * math.min(S.bhopSpeed, 60)
                if (tick() - S.bhopTimer) > 0.05 then
                    hum.Jump = true
                    S.bhopTimer = tick()
                end
            else
                if S.bhopBV then
                    S.bhopBV:Destroy(); S.bhopBV = nil
                end
                if hSpeed < 4 then S.bhopSpeed = 0 end
            end
        end
    else
        if S.bhopBV then S.bhopBV:Destroy(); S.bhopBV = nil end
        S.bhopSpeed = 0
        S.bhopAir = false
    end

    -- ── SilentAim target finding ─────────────
    S.saTarget = nil
    if (MOD.silentaim.enabled or MOD.autoshoot.enabled) then
        local mousePos = UserInputService:GetMouseLocation()
        local closestDist = MOD.silentaim.set.fov
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl ~= LP and pl.Character then
                local h = pl.Character:FindFirstChildOfClass("Humanoid")
                if h and h.Health > 0 then
                    local head = pl.Character:FindFirstChild("Head")
                    if head then
                        local sp, onScr = Cam:WorldToViewportPoint(head.Position)
                        if onScr then
                            local dist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
                            if dist < closestDist then
                                closestDist = dist; S.saTarget = head
                            end
                        end
                    end
                end
            end
        end
        if fovCircle and HAS_DRAW then
            if MOD.silentaim.enabled and MOD.silentaim.set.showFov then
                fovCircle.Position = mousePos
                fovCircle.Radius   = MOD.silentaim.set.fov
                fovCircle.Color    = S.saTarget
                    and Color3.fromRGB(255, 70, 70)
                    or  Color3.fromRGB(255, 255, 255)
                fovCircle.Visible  = true
            else
                fovCircle.Visible = false
            end
        end
    elseif fovCircle and HAS_DRAW then
        fovCircle.Visible = false
    end

    -- ── ESP ───────────────────────────────────
    if MOD.esp.enabled and HAS_DRAW then
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl ~= LP then
                local d = getOrCreateESP(pl)
                if d then
                    local c2  = pl.Character
                    local hrp2= c2 and c2:FindFirstChild("HumanoidRootPart")
                    local h2  = c2 and c2:FindFirstChildOfClass("Humanoid")
                    if hrp2 and h2 and h2.Health > 0 then
                        local center, onScr = Cam:WorldToViewportPoint(hrp2.Position)
                        if onScr then
                            local topSP, _ = Cam:WorldToViewportPoint(hrp2.Position + Vector3.new(0, 3.2, 0))
                            local bH = math.abs(topSP.Y - center.Y) * 2
                            local bW = bH * 0.5
                            local col = isMurd(pl) and Color3.fromRGB(255, 60, 60)
                                or (isSher(pl) and Color3.fromRGB(60, 120, 255)
                                or Color3.fromRGB(220, 220, 220))
                            local role = isMurd(pl) and " [M]" or (isSher(pl) and " [S]" or "")
                            d.box.Color = col; d.box.Visible = true
                            d.box.Size = Vector2.new(bW, bH)
                            d.box.Position = Vector2.new(center.X - bW/2, topSP.Y)
                            d.name.Text = pl.Name .. role
                            d.name.Color = col; d.name.Visible = true
                            d.name.Position = Vector2.new(center.X, topSP.Y - 15)
                        else
                            d.box.Visible = false; d.name.Visible = false
                        end
                    else
                        d.box.Visible = false; d.name.Visible = false
                    end
                end
            end
        end
    else
        for _, d in pairs(espData) do
            if d.box  then d.box.Visible  = false end
            if d.name then d.name.Visible = false end
        end
    end

    -- ── ShootTrails ───────────────────────────
    updateTrails()

    -- ── CsPov viewmodel ───────────────────────
    if MOD.cspov.enabled and vmFrame.Visible then
        updateVMBob(dt)
    end

    -- ── Watermark ────────────────────────────
    if MOD.watermark.enabled and wmFrame.Visible then
        local ping = 0
        pcall(function() ping = math.floor(LP:GetNetworkPing() * 1000) end)
        wmStatsLbl.Text = string.format("%d FPS  |  %dms", S.fps, ping)
    end

    -- ── Speed HUD ────────────────────────────
    if MOD.speedhud.enabled and speedFrame.Visible and hrp then
        local hVel = hrp.Velocity * Vector3.new(1, 0, 1)
        speedVal.Text = tostring(math.floor(hVel.Magnitude))
    end
end)
